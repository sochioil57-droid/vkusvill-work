📊 Подключение к Google Таблицам

Чтобы все поступающие анкеты автоматически записывались в вашу личную Google Таблицу:

  1. Создайте новую Google Таблицу (Google Sheets).
  2. В верхнем меню нажмите Расширения → Apps Script.
  3. Вставьте следующий код:
function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var p = e.parameter;
  sheet.appendRow([
    new Date(),
    p.brand || 'ВкусВилл',
    p.name || '',
    p.phone || '',
    p.vacancy || '',
    p.experience || '',
    p.lastJob || '',
    p.city || '',
    p.hours || '',
    p.days || '',
    p.estMonth || ''
  ]);
  return ContentService.createTextOutput("OK").setMimeType(ContentService.MimeType.TEXT);
}
